cmake_minimum_required(VERSION 3.15)
project(powerscope_npu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(ANLOGIC_ARMNN_ROOT "" CACHE PATH "Vendor face_detection/libs/armnn_lib directory")
if(NOT EXISTS "${ANLOGIC_ARMNN_ROOT}/include/armnn/ArmNN.hpp")
  message(FATAL_ERROR "Set ANLOGIC_ARMNN_ROOT to the extracted board SDK")
endif()
foreach(probe IN ITEMS npu_probe parser_probe tensor_io_test vendor_model_probe student_pipeline_probe ps_npu_worker)
  add_executable(${probe} ${probe}.cpp)
  target_include_directories(${probe} PRIVATE "${ANLOGIC_ARMNN_ROOT}/include")
  target_link_directories(${probe} PRIVATE "${ANLOGIC_ARMNN_ROOT}/lib")
  target_link_options(${probe} PRIVATE "LINKER:-rpath-link,${ANLOGIC_ARMNN_ROOT}/lib")
  target_link_libraries(${probe} PRIVATE armnnOnnxParser armnn protobuf pthread dl)
  set_target_properties(${probe} PROPERTIES BUILD_RPATH "${ANLOGIC_ARMNN_ROOT}/lib")
endforeach()

add_library(powerscope_mmio SHARED mmio.cpp)
